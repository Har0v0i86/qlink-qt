cmake_minimum_required(VERSION 3.16)
project(Qlink LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTOUIC ON)
set(CMAKE_AUTORCC ON)
find_package(Qt6 REQUIRED COMPONENTS Widgets)
set(GAME_SOURCES GameBoard.cpp GameBoard.h Character.cpp Character.h GameLogic.cpp GameLogic.h resource.qrc)
add_executable(Qlink main.cpp MainWindow.cpp MainWindow.h MainWindow.ui MainMenu.cpp MainMenu.h ${GAME_SOURCES})
target_link_libraries(Qlink PRIVATE Qt6::Widgets)
include(CTest)
if(BUILD_TESTING)
    find_package(Qt6 REQUIRED COMPONENTS Test)
    add_executable(tst_gameboard tests/tst_gameboard.cpp ${GAME_SOURCES})
    target_include_directories(tst_gameboard PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
    target_link_libraries(tst_gameboard PRIVATE Qt6::Test Qt6::Widgets)
    add_test(NAME gameboard COMMAND tst_gameboard)
    set_tests_properties(gameboard PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
endif()
